✨ SQL: Creating Database and Table ✨

1️⃣ Create a Database named "library"

CREATE DATABASE library;

2️⃣ Delete the "library" Database

DROP DATABASE library;

3️⃣ Create a Table "book_records" with Columns

CREATE TABLE book_records (
    Book_Library_No INT PRIMARY KEY,
    Book_Name VARCHAR(100),
    Author_Name VARCHAR(100),
    Book_Edition VARCHAR(50),
    Price DECIMAL(10, 2)
);

4️⃣ Delete the "book_records" Table

DROP TABLE book_records;
